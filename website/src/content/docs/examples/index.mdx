Each example works a real biological question through to a result with DBRetina, on real data. They are small
enough to run yourself, each ends in a figure, and each cites the sources behind the result.

import { Card, CardGrid } from '@astrojs/starlight/components';

	</Card>
	<Card title="Comorbidity, not cause: autism's genetic neighbours" icon="seti:plan">

	</Card>
	<Card title="One name, two mechanisms: type 1 and type 2 diabetes" icon="random">

		[Read it](/DBRetina/examples/diabetes-types/)
	</Card>
	<Card title="Alzheimer's disease and the gene-set size confound" icon="warning">
